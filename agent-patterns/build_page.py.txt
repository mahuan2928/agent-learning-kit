def code(path, title=None, open_=True, badge="源码"):
    text = src(path).rstrip("\n")
    lines = "".join(f'<span class="l"><i>{n}</i>{esc(l) or " "}</span>' for n, l in enumerate(text.split("\n"), 1))
    n = text.count("\n") + 1
    pre = f'<pre class="code"><code>{lines}</code></pre>'
    if open_:
        return (f'<div class="file"><div class="fh"><span class="badge new">{badge}</span><b>{esc(path)}</b>'
                f'<span class="meta">{n} 行</span></div>{pre}</div>')
    return (f'<div class="file"><div class="fh"><span class="badge mod">{badge}</span><b>{esc(path)}</b>'
            f'<span class="meta">{n} 行</span></div><details class="full"><summary>展开 {esc(path)}</summary>{pre}</details></div>')

def flow(spec):
    """spec: list of tokens. ('llm'|'code'|'human'|'n', text, small?) | '>' | '/' (line break) | ('side', text)"""
    out = []
    for t in spec:
        if t == ">": out.append('<span class="ar">→</span>')
        elif t == "/": out.append('<span class="br"></span>')
        elif t[0] == "side": out.append(f'<span class="side">{t[1]}</span>')
        else:
            k, txt, *sm = t
            out.append(f'<span class="n {k}">{txt}' + (f'<small>{sm[0]}</small>' if sm else "") + '</span>')
    return f'<div class="flow">{"".join(out)}</div>'

  <h3>组合：真实系统很少只用一个模式</h3>
  <p>客服系统 = <strong>P2</strong> 分拣 → FAQ 类走 <strong>P6</strong> 检索回答 → 对外回复过 <strong>P5</strong> 审查。提案生成 = <strong>P1</strong> 链条，其中「方案章节」那一步是 <strong>P4</strong>（按课题数动态拆分），终稿过 <strong>P3</strong> 多视角审查。面试时用组合来讲，比逐个背定义有说服力得多。</p>

  <h3>面试里最常被追问的五个问题</h3>
  <div class="qa"><span class="lbl">通用追问</span><div class="body"><dl>
    <dt>什么时候该用 agent，什么时候用 workflow？</dt>
    <dd>步骤能事先写出来就用 workflow：便宜、可预测、好测。只有「下一步做什么要看上一步的结果」而且步数不定时，才把那一段交给自主循环。即使用了 agent，也要用代码设上限（轮数、预算、工具子集）。</dd>
    <dt>怎么决定哪些交给模型、哪些交给代码？</dt>
    <dd>能写成确定规则的（字数、禁用词、算术、权限、出处是否存在）一律代码；需要理解语言的（分类、改写、生成、按 rubric 判断）交给模型。高风险判断就算需要理解语言，也要加规则兜底，比如 P2 的健康被害关键词。</dd>
    <dt>模型怎么选？</dt>
    <dd>按节点选：路由、投票、批量 worker 用 Haiku；编排、生成、裁判用 Opus。先用最强的模型把质量做出来，再在评测集上逐个节点降级，看质量掉不掉。不要一开始就为了省钱用小模型。</dd>
    <dt>成本怎么控？</dt>
    <dd>看每个任务的总成本，不看单次调用。手段按顺序是：缓存（免费）→ 调低 effort → 便宜模型做简单节点 → 用代码检查提前截断（P1 的 Gate、P5 的机械检查）→ 离线任务走 Batches。</dd>
    <dt>怎么防 prompt injection？</dt>
    <dd>外部内容一律放进标签并声明是数据；高风险动作要人确认；最关键的是最小权限：模型拿不到的数据和工具，被注入了也做不了什么。P2 的 FAQ 回复步骤只拿到 FAQ 文本，就是这个原则。</dd>
